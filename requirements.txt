# Direct runtime dependencies only. Transitive packages are deliberately left
# for pip to resolve.
#
# This file used to be `pip freeze` output, which pinned the entire dependency
# closure of the old google-generativeai SDK. Migrating to google-genai left
# those orphans behind, and one of them -- google-auth==2.55.0 -- contradicted
# google-genai's own floor of >=2.56.0, so a clean install could not resolve at
# all. It kept working locally because the venv was already populated; CI
# resolves from scratch every run and failed in 20 seconds.
#
# Pin what the code imports. Let pip solve the rest.

# Gemini API client (src/tutor.py, src/lesson_engine.py).
google-genai==2.17.0

# Loads GEMINI_API_KEY from .env (src/tutor.py).
python-dotenv==1.2.2

# Image handling across capture, OCR and highlighting.
pillow==12.2.0

# OCR. Also needs the Tesseract binary installed separately -- see readme.
pytesseract==0.3.13

# Screen capture (src/capture/screen_capture.py, src/desktop/capture.py).
mss==9.0.1

# Frame buffers for the recorder and array maths in the benchmark.
numpy==2.4.6

# Demo recording (src/desktop/recorder.py, tools/record_demo.py).
imageio==2.36.1
imageio-ffmpeg==0.5.1

# Desktop overlay UI. Required by desktop.py and the whole src/attention,
# src/desktop and src/input layer -- the primary run mode.
PyQt6==6.11.0

# Global hotkey capture (src/input/hotkeys.py). Needs root on Linux.
keyboard==0.13.5

# Browser front-end (app.py). Not needed for the desktop overlay.
streamlit==1.58.0
